# ---------------- HELPERS ----------------
def new_item(name, cat, rar, eff, qty=1, val=0):
    return {
        "id": f"{random.randint(10000,99999)}",
        "name": name, "category": cat, "rarity": rar,
        "effect": eff, "quantity": qty, "value": val,
        "equipped": False,
    }

def random_item():
    cat = random.choice(CATEGORIES)
    name, base, eff = random.choice(TEMPLATES.get(cat, [("Item", 10, "-")]))
    rar = random.choices(list(RARITIES.keys()), weights=[45,25,15,9,5,1])[0]
    val = int(base * RARITIES[rar])
    return new_item(name, cat, rar, eff, 1, val)

# ---------------- CENTRAL DE ITENS ----------------

    st.header("🎒 Central de Itens")
    sub = st.tabs(["Inventário", "Equipamento", "Loja", "Baú", "Combate"])

    # INVENTÁRIO
    with sub[0]:
        c1, c2, c3, c4 = st.columns(4)
        with c1:

                if len(st.session_state.inv) < st.session_state.cap:

        with c2:
            if st.button("💰 +50 Ouro"):
                st.session_state.gold += 50
                st.rerun()
        with c3:
            if st.button("📦 Expandir (+10)"):
                if st.session_state.gold >= 100:
                    st.session_state.gold -= 100
                    st.session_state.cap += 10
                    st.rerun()
        with c4:
            if st.button("🗑️ Limpar"):
                st.session_state.inv = []
                st.rerun()

        st.divider()

        if not st.session_state.inv:
            st.info("Inventário vazio.")
        else:
            cols = st.columns(3)
            for i, item in enumerate(st.session_state.inv):
                with cols[i % 3]:
                    eq = " ⚔️" if item["equipped"] else ""
                    st.markdown(f"""<div class="item-card item-{item['rarity']}">
                    <div class="item-name">{item['name']}{eq}</div>
                    <div class="item-meta">{item['category']} · {item['rarity']} · x{item['quantity']} · {item['value']}💰</div>
                    <div class="item-desc">{item['effect']}</div>
                    </div>""", unsafe_allow_html=True)

                    b1, b2, b3 = st.columns(3)
                    with b1:
                        if item["category"] in SLOTS:
                            if item["equipped"]:
                                if st.button("Desc.", key=f"d{i}"):
                                    st.session_state.equip[item["category"]] = None
                                    item["equipped"] = False
                                    st.rerun()
                            else:
                                if st.button("Equip.", key=f"e{i}"):
                                    old = st.session_state.equip.get(item["category"])
                                    if old: old["equipped"] = False
                                    st.session_state.equip[item["category"]] = item
                                    item["equipped"] = True
                                    st.rerun()
                    with b2:
                        if item["category"] == "Consumível":
                            if st.button("Usar", key=f"u{i}"):
                                if item["quantity"] > 1: item["quantity"] -= 1
                                else: st.session_state.inv.remove(item)
                                st.session_state.stats["HP"] = min(
                                    st.session_state.stats["HPmax"],
                                    st.session_state.stats["HP"] + 50
                                )
                                st.rerun()
                    with b3:
                        if st.button("Vender", key=f"v{i}"):
                            st.session_state.gold += int(item["value"] * 0.5) * item["quantity"]
                            st.session_state.inv.remove(item)
                            st.rerun()

    # EQUIPAMENTO
    with sub[1]:
        st.subheader("⚔️ Slots")
        cols = st.columns(4)
        for i, slot in enumerate(SLOTS):
            with cols[i % 4]:
                it = st.session_state.equip.get(slot)
                if it:
                    st.markdown(f"""<div class="slot-filled">
                    <div class="item-meta">{slot}</div>
                    <div class="item-name">{it['name']}</div>
                    <div class="item-desc">{it['effect']}</div>
                    </div>""", unsafe_allow_html=True)
                    if st.button("Desequipar", key=f"ds{i}"):
                        it["equipped"] = False
                        st.session_state.equip[slot] = None
                        st.rerun()
                else:
                    st.markdown(f'<div class="slot-empty">◯ {slot}<br>vazio</div>', unsafe_allow_html=True)

    # LOJA
    with sub[2]:
        st.subheader("🏪 Loja")
        if st.button("🔄 Renovar Estoque (20💰)"):
            if st.session_state.gold >= 20:
                st.session_state.gold -= 20
                st.session_state.shop = [random_item() for _ in range(6)]
                st.rerun()
        st.divider()
        cols = st.columns(3)
        for i, item in enumerate(st.session_state.shop):
            with cols[i % 3]:
                st.markdown(f"""<div class="item-card item-{item['rarity']}">
                <div class="item-name">{item['name']}</div>
                <div class="item-meta">{item['category']} · {item['rarity']} · {item['value']}💰</div>
                <div class="item-desc">{item['effect']}</div>
                </div>""", unsafe_allow_html=True)
                if st.button(f"Comprar ({item['value']}💰)", key=f"b{i}"):
                    if st.session_state.gold >= item["value"] and len(st.session_state.inv) < st.session_state.cap:
                        st.session_state.gold -= item["value"]
                        st.session_state.inv.append(item)
                        st.session_state.shop.remove(item)
                        st.rerun()

    # BAÚ
    with sub[3]:
        st.subheader("🔐 Baú")
        c1, c2 = st.columns(2)
        with c1:
            st.write(f"Inventário: {len(st.session_state.inv)} itens")
            if st.session_state.inv:
                names = [f"{x['name']} ({x['rarity']})" for x in st.session_state.inv]
                sel = st.selectbox("Enviar para o baú", names, key="si")
                if st.button("→ Enviar"):
                    it = st.session_state.inv[names.index(sel)]
                    st.session_state.inv.remove(it)
                    st.session_state.stash.append(it)
                    st.rerun()
        with c2:
            st.write(f"Baú: {len(st.session_state.stash)} itens")
            if st.session_state.stash:
                names2 = [f"{x['name']} ({x['rarity']})" for x in st.session_state.stash]
                sel2 = st.selectbox("Retirar do baú", names2, key="so")
                if st.button("← Retirar"):
                    if len(st.session_state.inv) < st.session_state.cap:
                        it = st.session_state.stash[names2.index(sel2)]
                        st.session_state.stash.remove(it)
                        st.session_state.inv.append(it)
                        st.rerun()

    # COMBATE
    with sub[4]:
        st.subheader("⚔️ Simulador de Combate")
        st.write(f"**Nível {st.session_state.level}** · ATK {st.session_state.stats['ATK']} · DEF {st.session_state.stats['DEF']}")
        if st.button("⚔️ Lutar!", type="primary"):
            enemy_hp = random.randint(30, 80)
            enemy_atk = random.randint(5, 15)
            my_hp = st.session_state.stats["HP"]
            log = []
            r = 0
            while my_hp > 0 and enemy_hp > 0 and r < 15:
                r += 1
                dmg = max(1, st.session_state.stats["ATK"] - random.randint(0, 5))
                enemy_hp -= dmg
                log.append(f"R{r}: Você → {dmg} dano (Inimigo {max(0,enemy_hp)} HP)")
                if enemy_hp <= 0: break
                edmg = max(1, enemy_atk - st.session_state.stats["DEF"] // 2)
                my_hp -= edmg
                log.append(f"R{r}: Inimigo → {edmg} dano (Você {max(0,my_hp)} HP)")
            st.session_state.stats["HP"] = max(0, my_hp)
            if enemy_hp <= 0:
                gold = random.randint(20, 60)
                xp = random.randint(15, 40)
                st.session_state.gold += gold
                st.session_state.xp += xp
                while st.session_state.xp >= 100:
                    st.session_state.xp -= 100
                    st.session_state.level += 1
                    st.session_state.stats["HPmax"] += 10
                    st.session_state.stats["HP"] = st.session_state.stats["HPmax"]
                st.success(f"🏆 Vitória! +{gold}💰 +{xp}XP")
            else:
                st.error("💀 Derrota!")
            with st.expander("Ver log"):
                for linha in log:
                    st.write(linha)

st.divider()
